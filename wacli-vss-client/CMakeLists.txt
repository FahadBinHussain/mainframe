cmake_minimum_required(VERSION 3.16)
project(wacli_vss_client LANGUAGES CXX)
add_library(wacli_vss_client STATIC src/orchestrator.cpp src/native_vss_backend.cpp src/native_snapshot_copy.cpp)
target_compile_features(wacli_vss_client PUBLIC cxx_std_17)
target_include_directories(wacli_vss_client PUBLIC include)
if(WIN32)
    target_link_libraries(wacli_vss_client PRIVATE vssapi ole32 uuid advapi32)
    add_executable(wacli_vss_backup src/backup_copy_main.cpp)
    target_link_libraries(wacli_vss_backup PRIVATE wacli_vss_client)
    if(MINGW)
        target_link_options(wacli_vss_backup PRIVATE -municode)
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(vss_fake_tests tests/fake_tests.cpp)
    target_link_libraries(vss_fake_tests PRIVATE wacli_vss_client)
    add_test(NAME vss_fake_tests COMMAND vss_fake_tests)
    add_executable(vss_copy_tests tests/copy_tests.cpp)
    target_link_libraries(vss_copy_tests PRIVATE wacli_vss_client)
    add_test(NAME vss_copy_tests COMMAND vss_copy_tests)
    if(NOT WIN32)
        # Test-only headers execute actual adapters against memory-only API doubles.
        # Never used by the production library or Windows cross-compile targets.
        add_executable(vss_native_copy_fake_tests src/native_snapshot_copy.cpp tests/native_copy_tests.cpp)
        add_executable(vss_native_backend_fake_tests src/native_vss_backend.cpp src/orchestrator.cpp tests/native_backend_tests.cpp)
        foreach(target vss_native_copy_fake_tests vss_native_backend_fake_tests)
            target_compile_features(${target} PRIVATE cxx_std_17)
            target_compile_definitions(${target} PRIVATE _WIN32)
            target_include_directories(${target} PRIVATE tests/win32_fake include)
            add_test(NAME ${target} COMMAND ${target})
        endforeach()
    endif()
    if(WIN32)
        # Linkage evidence only: no VSS operation or source-root argument.
        add_executable(vss_link_only tests/link_only.cpp)
        target_link_libraries(vss_link_only PRIVATE wacli_vss_client)
    endif()
endif()
