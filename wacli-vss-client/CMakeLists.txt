cmake_minimum_required(VERSION 3.16)
project(wacli_vss_client LANGUAGES CXX)
add_library(wacli_vss_client STATIC src/orchestrator.cpp src/native_vss_backend.cpp)
target_compile_features(wacli_vss_client PUBLIC cxx_std_17)
target_include_directories(wacli_vss_client PUBLIC include)
if(WIN32)
    target_link_libraries(wacli_vss_client PRIVATE vssapi ole32 uuid)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(vss_fake_tests tests/fake_tests.cpp)
    target_link_libraries(vss_fake_tests PRIVATE wacli_vss_client)
    add_test(NAME vss_fake_tests COMMAND vss_fake_tests)
    if(WIN32)
        # Linkage evidence only: no VSS operation or source-root argument.
        add_executable(vss_link_only tests/link_only.cpp)
        target_link_libraries(vss_link_only PRIVATE wacli_vss_client)
    endif()
endif()
